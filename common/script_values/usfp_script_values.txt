usfp_calculate_banana_wars_score = {
    value = 0
    every_state = {
        limit = {
            is_in_geographic_region = geographic_region_caribbean_coast
            owner ?= {
                has_revolution = no
                NOT = {
                    this = root
                }
            }
        }
        owner ?= {
            #point: has investment rights or market
            if = {
                limit = {
                    OR = {
                        is_in_customs_union_with = root
                        any_scope_treaty = {
                            binds = root
                            any_scope_article = {
                                has_type = foreign_investment_rights
                                target_country = root
                            }
                        }
                    }
                }
                add = 1
            }
            #point: is subject
            if = {
                limit = {
                    is_subject_of = root
                }
                add = 1
            }
            #point: is non-democratic with Laissez Faire or Extraction Economy
            if = {
                limit = {
                    OR = {
                        is_subject_of = root
                        is_in_customs_union_with = root
                        any_scope_treaty = {
                            binds = root
                            any_scope_article = {
                                has_type = foreign_investment_rights
                                target_country = root
                            }
                        }
                    }
                    OR = {
                        has_law_or_variant = law_type:law_laissez_faire
                        has_law_or_variant = law_type:law_extraction_economy
                    }
                    OR = {
                        has_law_or_variant = law_type:law_autocracy
                        has_law_or_variant = law_type:law_oligarchy
                        has_law_or_variant = law_type:law_technocracy
                        has_law_or_variant = law_type:law_single_party_state
                    }
                }
                add = 1
            }
            #lose points for council republics
            if = {
                limit = {
                    has_law_or_variant = law_type:law_council_republic
                }
                subtract = 2
            }
        }
    }
    #point for each unincorporated state
    every_scope_state = {
        limit = {
            OR = {
                region = sr:region_central_america
                is_in_geographic_region = geographic_region_caribbean_old
                region = sr:region_gran_colombia
            }
            is_incorporated = no
            is_treaty_port = no
        }
        add = 1
    }

    max = 30
    min = 0
}

usfp_get_roosevelt_domestic_score = {
    value = 0
    root = {
        if = {
            limit = {
                usfp_roosevelt_agenda_national_parks_trigger = yes
            }
            add = 1
        }
        if = {
            limit = {
                has_variable = busted_trusts
            }
            add = 1
        }
        if = {
            limit = {
                has_law_or_variant = law_type:law_interventionism
            }
            add = 1
        }
        if = {
            limit = {
                institution_investment_level = {
                    institution = institution_workplace_safety
                    value >= 3
                }
            }
            add = 1
        }
        if = {
            limit = {
                institution_investment_level = {
                    institution = institution_health_system
                    value >= 3
                }
            }
            add = 1
        }
        if = {
            limit = {
                NOT = {
                    has_variable = roosevelt_didnt_avoid_revolution
                }
            }
            add = 1
        }
        if = {
            limit = {
                NOT = {
                    any_political_movement = {
                        movement_is_socialist = yes
                        OR = {
                            political_movement_radicalism > 0.75
                            political_movement_support > 0.15
                        }
                    }
                }
            }
            add = 1
        }
        if = {
            limit = {
                ig:ig_industrialists ?= {
                    is_powerful = no
                }
            }
            add = 1
        }
        #halved if you go for a third term
        if = {
            limit = {
                has_variable = roosevelt_went_third_term
            }
            divide = 2
        }
    }
    min = 0
}

usfp_get_roosevelt_diplomacy_score = {
    value = 0
    root = {
        if = {
            limit = {
                #Cuba
                OR = {
                    c:CUB ?= {
                        is_subject_of = root
                    }
                    AND = {
                        country_or_subject_owns_entire_state_region = STATE_CENTRAL_CUBA
                        country_or_subject_owns_entire_state_region = STATE_EASTERN_CUBA
                        country_or_subject_owns_entire_state_region = STATE_WESTERN_CUBA
                    }
                }
            }
            add = 1
        }
        if = {
            limit = {
                #Puerto Rico
                country_or_subject_owns_entire_state_region = STATE_PUERTO_RICO
            }
            add = 1
        }
        if = {
            limit = {
                #Panama Canal
                any_scope_building = {
                    is_building_type = building_panama_canal
                    is_under_construction = no
                }
            }
            add = 1
        }
        if = {
            limit = {
                #Monroe Doctrine enforced minus GBR
                NOT = {
                    any_country_in_americas = {
                        is_subject = yes
                        overlord ?= {
                            NOR = {
                                is_in_geographic_region = geographic_region_americas
                                c:GBR ?= this
                            }
                        }
                    }
                }
            }
            add = 1
        }
        if = {
            limit = {
                #Pacific Power Projection
                OR = {
                    any_subject_or_below = {
                        is_in_geographic_region = geographic_region_western_pacific
                        OR = {
                            is_subject_type = subject_type_colony
                            is_subject_type = subject_type_puppet
                        }
                    }
                    any_scope_state = {
                        is_in_geographic_region = geographic_region_western_pacific
                        count >= 3
                    }
                }
            }
            add = 1
        }
        if = {
            limit = {
                #Expanded Power Bloc
                AND = {
                    is_power_bloc_leader = yes
                    num_power_bloc_members_script > 7
                    power_bloc.current_cohesion_percentage >= 0.90
                    power_bloc.power_bloc_rank >= 3
                }
            }
            add = 1
        }
        if = {
            limit = {
                #Army Reform
                AND = {
                    any_scope_building = {
                        filter = {
                            is_building_type = building_barrack
                        }
                        percent = 1
                        OR = {
                            has_active_production_method = pm_advanced_tactics_training
                            has_active_production_method = pm_training_streamlining
                            has_active_production_method = pm_nco_incorporation
                            has_active_production_method = pm_mobile_warfare_tactics
                        }
                    }
                    army_power_projection >= var:roosevelt_army_target
                }
            }
            add = 1
        }
        if = {
            limit = {
                #Complete Great White Fleet
                OR = {
                    has_variable = new_navy_completed
                    naval_power_projection >= root.var:roosevelt_navy_target
                }
            }
            add = 1
        }

        #halved if you go for a third term
        if = {
            limit = {
                has_variable = roosevelt_went_third_term
            }
            divide = 2
        }
    }

usfp_second_revolution_get_americanism_target = {
	# Primary Cultures Fervor
	add = {
		every_primary_culture = {
			add = culture_current_fervor
		}
        divide = {
            every_primary_culture = {
                add = 1
            }
        }
        divide = 100
		desc = "average_primary_cultures_fervor"
	}
	#Manifest Destiny points
	if = {
		limit = {
			has_journal_entry = usfp_manifest_destiny_main
			has_variable = manifest_destiny_var
		}
		add = {
			value = var:manifest_destiny_var
            divide = 13
			desc = "manifest_destiny"
		}
	}
	# Completed Superculture
	if = {
		limit = {
			country_has_primary_culture = cu:usfp_american
		}
		add = {
			value = 1
			desc = "formed_american_superculture"
		}
	}
	# Supremacist and Traditionalist Movement
	add = {
		every_political_movement = {
			limit = {
				OR = {
					is_political_movement_type = movement_cultural_majority
					is_political_movement_type = movement_religious_majority
				}
			}
			add = political_movement_radicalism
			add = political_movement_support
		}
		desc = "supremacist_and_traditionalist_activism_and_support"
	}
	# Constitutionalist and Liberal Movement
	add = {
		every_political_movement = {
			limit = {
				OR = {
					is_political_movement_type = movement_liberal
					is_political_movement_type = movement_usfp_american_constitution
					is_political_movement_type = movement_usfp_progressive
				}
			}
			add = political_movement_radicalism
			add = political_movement_support
		}
		desc = "liberal_progressive_constitutionalist_activism_and_support"
	}
	# Fascist and White Movement
	add = {
		every_political_movement = {
			limit = {
				OR = {
					is_political_movement_type = movement_fascist
					is_political_movement_type = movement_usfp_neoconfederate
					is_political_movement_type = movement_white
				}
			}
			add = political_movement_radicalism
			add = political_movement_support
		}
		desc = "fascist_neoconfederate_white_activism_and_support"
	}
	# IGs supporting Presidential Republic
	add = {
		every_interest_group = {
			limit = {
				ig_counts_as_marginal = no
				law_enactment_stance = {
					value >= approve
					law = law_type:law_presidential_republic
				}
			}
			add = ig_clout
		}
		desc = "igs_supporting_presidential_republic"
	}
	# American Laws
	if = {
		limit = {
			has_law = law_type:law_usfp_nominal_separation
		}
		add = {
			value = 1
			desc = "law_usfp_nominal_separation"
		}
	}
	if = {
		limit = {
			has_law_or_variant = law_type:law_interventionism
		}
		add = {
			value = 0.5
			desc = "law_interventionism"
		}
	}
	if = {
		limit = {
			has_law_or_variant = law_type:law_laissez_faire
		}
		add = {
			value = 1
			desc = "law_interventionism"
		}
	}
	if = {
		limit = {
			has_variable = usfp_second_revolution_supported_americanism
		}
		add = {
			add = institution_investment_level:institution_schools
			add = institution_investment_level:institution_home_affairs
			divide = 5
			min = 0
			desc = "institutions_support_americanism"
		}
	}
    #Consolidation in State and Revolution
}

usfp_second_revolution_get_antiamericanism_target = {

			}
			add = political_movement_radicalism

				}
			}

		}
		desc = "communist_movement_activism_and_support"
	}

	add = {
		every_political_movement = {
			limit = {
				OR = {
					is_political_movement_type = movement_cultural_minority
					is_political_movement_type = movement_minority_rights
				}
			}
			add = political_movement_radicalism
			add = political_movement_support
		}
		desc = "minority_movements_activism_and_support"
	}

	add = {
		value = 1
		subtract = {
			every_primary_culture = {
                save_temporary_scope_as = primary_culture_scope
				add = "root.culture_percent_country(scope:primary_culture_scope)"
			}
		}
		desc = "percent_non_primary_cultures"
	}
	# Non-Homeland Incorporated States
	add = {
		every_scope_state = {
			limit = {
				is_incorporated = yes
				NOT = {
					is_homeland_of_country_cultures = root
				}
			}
			add = 0.2
		}
		desc = "non_homeland_incorporate_states"
	}
	# IGs supporting Council Republic
	add = {
		every_interest_group = {
			limit = {
				ig_counts_as_marginal = no
				law_stance = {
					value >= approve
					law = law_type:law_council_republic
				}
			}
			add = ig_clout
		}
		desc = "igs_supporting_council_republic"
	}
	# Revolutionary Laws
	if = {
		limit = {
			has_law_or_variant = law_type:law_state_atheism
		}
		add = {
			value = 1
			desc = "law_state_atheism"
		}
	}
	if = {
		limit = {
			has_law_or_variant = law_type:law_multicultural
		}
		add = {
			value = 0.5
			desc = "law_multicultural"
		}
	}
	if = {
		limit = {
			has_law_or_variant = law_type:law_anarchy
		}
		add = {
			value = 1
			desc = "law_anarchy"
		}
	}
	if = {
		limit = {
			has_law_or_variant = law_type:law_no_colonial_affairs
		}
		add = {
			value = 1
			desc = "law_no_colonial_affairs"
		}
	}
	if = {
		limit = {
			has_law_or_variant = law_type:law_single_party_state
		}
		add = {
			value = 1
			desc = "law_single_party_state"
		}
	}
	if = {
		limit = {
			country_has_voting_franchise = no
			NOT = {
				has_law_or_variant = law_type:law_anarchy
			}
		}
		add = {
			value = 1
			desc = "no_voting_franchise"
		}
	}
	if = {
		limit = {
			has_variable = usfp_second_revolution_supported_antiamericanism
		}
		add = {
			add = institution_investment_level:institution_schools
			add = institution_investment_level:institution_home_affairs
			divide = 5
			min = 0
			desc = "institutions_support_antiamericanism"
		}
	}
	#Monument Destruction
	if = {
		limit = {
			has_variable = usfp_destroyed_american_monuments
		}
		add = {
			value = 3
			desc = "destroyed_american_monuments"
		}
	}
    #At war with Ethnostate
    if = {
        limit = {
            is_at_war = yes
            any_enemy_in_war = {
                country_rank >= rank_value:major_power
                has_law_or_variant = law_type:law_ethnostate
            }

            value = 0.5
            desc = "at_war_with_ethnostate"
        }
    }
