usfp_take_bribes_button = {
	name = "usfp_take_bribes_button"
	desc = "usfp_take_bribes_button_desc"

	visible = {
		NOT = {
			ruler = {
				has_trait = honorable
			}
		}
	}
	
	possible = {
		any_scope_character = {
			interest_group = {
				is_in_government = yes
			}
			has_role_of_type = politician
			NOR = {
				has_trait = grifter
				has_trait = honorable
				has_ideology = ideology:ideology_usfp_progressive
				has_ideology = ideology:ideology_anarchist
				has_ideology = ideology:ideology_communist
				has_ideology = ideology:ideology_luddite
			}
		}
		NOT = {
			has_modifier = usfp_federal_corruption_modifier
		}
	}

	effect = {
		random_scope_character = {
			limit = {
				interest_group = {
					is_in_government = yes
				}
				has_role_of_type = politician
				NOR = {
					has_trait = grifter
					has_trait = honorable
					has_ideology = ideology:ideology_usfp_progressive
					has_ideology = ideology:ideology_anarchist
					has_ideology = ideology:ideology_communist
					has_ideology = ideology:ideology_luddite
				}
			}
			add_trait = grifter
		}
		add_modifier = {
			name = usfp_federal_corruption_modifier
			years = 4
		}
		ig:ig_industrialists = {
			add_modifier = {
				name = usfp_industrialists_corruption_modifier
				years = 4
			}
		}
		random_list = {
			10 = {}
			30 = {
				add_loyalists = {
					value = small_radicals
					pop_type = capitalists
				}
			}
			30 = {
				add_loyalists = {
					value = small_radicals
					pop_type = capitalists
				}
				add_radicals = {
					value = small_radicals
					pop_type = bureaucrats
				}
			}
			30 = {
				add_loyalists = {
					value = small_radicals
					pop_type = capitalists
				}
				add_radicals = {
					value = small_radicals
					strata = lower
				}
			}
		}
	}
}

usfp_consolidate_trust_button = {
	name = "usfp_consolidate_trust_button"
	desc = "usfp_consolidate_trust_button_desc"

	visible = {
		NOT = {
			has_variable = consolidated_trust
		}
	}

	possible = {
		ig:ig_industrialists = {
			is_in_government = yes
		}
	}

	effect = {
		set_variable = consolidated_trust
		je:usfp_gilded_age = {
			add_modifier = {
				name = usfp_trust_dominance_modifier
			}
		}

		add_radicals = {
			value = small_radicals
			pop_type = shopkeepers
		}
		add_loyalists = {
			value = small_radicals
			pop_type = capitalists
		}
		ig:ig_industrialists = {
			add_modifier = {
				name = usfp_industrialists_corruption_modifier
				years = 4
			}
		}
	}
}

usfp_nationalize_railroads_button = {
	name = "usfp_nationalize_railroads_button"
	desc = "usfp_nationalize_railroads_button_desc"

	visible = {
		NOT = {
			has_variable = usfp_nationalized_railroads
		}
	}

	possible = {
		NOT = {
			has_law_or_variant = law_type:law_laissez_faire
		}
		any_scope_building = {
			filter = {
				is_building_type = building_railway
			}
			percent >= 0.75
			country_ownership_fraction >= 0.75
		}
	}

	effect = {
		custom_tooltip = {
			text = prevent_railroad_crash
			set_variable = usfp_nationalized_railroads
		}
		every_interest_group = {
			limit = {
				OR = {
					is_interest_group_type = ig_industrialists
					law_stance = {
						law = law_type:law_laissez_faire
						value >= approve
					}
				}
				law_stance = {
					law = law_type:law_command_economy
					value <= neutral
				}
			}
			add_modifier = {
				name = usfp_ig_nationalized_railways_negative_modifier
				years = 5
			}
		}
		every_interest_group = {
			limit = {
				OR = {
					leader = {
						has_ideology = ideology:ideology_usfp_progressive
					}
					law_stance = {
						law = law_type:law_command_economy
						value >= approve
					}
				}
			}
			add_modifier = {
				name = usfp_ig_nationalized_railways_positive_modifier
				years = 5
			}
		}
	}
}
