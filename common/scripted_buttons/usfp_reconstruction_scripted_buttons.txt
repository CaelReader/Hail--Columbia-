#Enforce Voting Rights
usfp_reconstruction_enforce_rights_button = {
	name = "usfp_reconstruction_enforce_rights_button"
	desc = "usfp_reconstruction_enforce_rights_button_desc"

	visible = {
		NOT = {
			has_variable = reconstruction_enforced_voting_rights
		}
	}

	possible = {

		usfp_is_afro_american_discriminated = no
		authority >= 150
		OR = {
			institution_investment_level = {
				institution = institution_police
				value >= 3
			}
			institution_investment_level = {
				institution = institution_home_affairs
				value >= 2
			}
		}
		usfp_has_dixie_occupation_trigger = yes
	}

	effect = {
		set_variable = reconstruction_enforced_voting_rights
		add_radicals = {
			value = medium_radicals
			culture = cu:dixie
		}
		add_loyalists = {
			value = medium_radicals
			culture = cu:afro_american
		}
		usfp_reconstruction_swing_freedmen = yes
		hidden_effect = {
			usfp_reconstruction_swing_freedmen = yes
			usfp_reconstruction_swing_freedmen = yes
		}
		add_modifier = {
			name = usfp_enforcing_reconstruction
			multiplier = 2
			years = 3
		}
		usfp_destabilize_union_effect = {MULT = 1}
		usfp_anger_slavers_effect = {MULT = 1}
	}

	ai_chance = {
		value = 5
	}
}

#Amnesty for Confederates
usfp_reconstruction_confederate_amnesty_button = {
	name = "usfp_reconstruction_confederate_amnesty_button"
	desc = "usfp_reconstruction_confederate_amnesty_button_desc"

	visible = {
		NOT = {
			has_variable = reconstruction_confederate_amnesty
		}
	}

	possible = {
		ruler = {
			NOR = {
				has_ideology = ideology:ideology_abolitionist
				has_ideology = ideology:ideology_usfp_abolitionist_radical
			}
		}
	}

	effect = {
		set_variable = reconstruction_confederate_amnesty
		usfp_reconstruction_swing_south_severe = yes
		add_loyalists = {
			value = medium_radicals
			culture = cu:dixie
		}
		add_loyalists = {
			value = medium_radicals
			culture = cu:dixie
			strata = upper
		}
		add_radicals = {
			value = medium_radicals
			culture = cu:afro_american
		}
		add_radicals = {
			value = small_radicals
			culture = cu:yankee
		}
		usfp_appease_slavers_effect = {MULT = 2}
		every_interest_group = {
			limit = {
				has_modifier = revolution_lost_ig
			}
			remove_modifier = revolution_lost_ig
		}
		every_interest_group = {
			limit = {
				NOR = {
					has_ideology = ideology:ideology_anti_slavery
					leader = { usfp_anti_slavery_leader_trigger = yes }
					has_ideology = ideology:ideology_pro_slavery
					leader = { has_ideology = ideology:ideology_slaver }
				}
			}
			add_modifier = {
				name = usfp_healing_the_nation_modifier
				years = 4
			}
		}
	}

	ai_chance = {
		value = 25
		if = {
			limit = {
				usfp_anti_abolitionist_government = yes
			}
			add = 100
		}
	}
}

#40 Acres and a Mule
usfp_reconstruction_radical_land_distribution_button = {
    name = "usfp_reconstruction_radical_land_distribution_button"
    desc = "usfp_reconstruction_radical_land_distribution_button_desc"

    visible = {
        any_scope_state = {
            # has_modifier = unreconstructed_state
            state_region = {
                is_homeland = cu:afro_american
            }
            NOT = {
                has_modifier = usfp_land_redistribution
            }
        }
        NOT = {
            any_scope_state = {
                has_decree = decree_usfp_radical_land_distribution
            }
        }
        NOT = {
            any_active_law = {
                has_amendment = amendment_type:amendment_usfp_radical_land_redistribution
            }
        }
    }

    possible = {
        usfp_has_dixie_occupation_trigger = yes
        custom_tooltip = {
            text = radical_abolitionist_government
            any_interest_group_in_government = {
                leader = {
                    OR = {
                        has_ideology = ideology:ideology_usfp_abolitionist_radical
                        has_ideology = ideology:ideology_humanitarian
                    }
                }
            }
        }
    }

    effect = {
        add_radicals = {
            value = very_large_radicals
            culture = cu:dixie
            strata = upper
        }
        active_law:lawgroup_slavery ?= {
            add_amendment = {
                type = amendment_usfp_radical_land_redistribution
                sponsor = scope:radical_abolitionist_ig
            }
        }
        custom_tooltip = {
            text = acw_followup_events.4.ctt
        }
        if = {
            limit = {
                NOT = {
                    exists = scope:radical_abolitionist_ig
                }
            }
            random_interest_group_in_government = {
                limit = {
                    leader = {
                        OR = {
                            has_ideology = ideology:ideology_usfp_abolitionist_radical
                            has_ideology = ideology:ideology_humanitarian
                        }
                    }
                }
                save_scope_as = radical_abolitionist_ig #in case there wasn't a fitting leader when the event popped
            }
        }
    }

    ai_chance = {
        value = 0
        if = {
            limit = {
                ruler ={
                    OR = {
                        has_ideology = ideology:ideology_usfp_abolitionist_radical
                        has_ideology = ideology:ideology_humanitarian
                    }
                }
            }
            add = 25
        }
    }
}
