#Coal Wars
usfp_je_coal_wars = {

    icon = gfx/interface/icons/goods_icons/coal.dds

    group = je_group_historical_content

    scripted_progress_bar = usfp_coal_wars_union_leverage_bar

    modifiers_while_active = {
        usfp_coal_wars_ongoing_strikes
    }

    is_shown_when_inactive = {
        c:USA ?= root
        NOT = {
            has_law_or_variant = law_type:law_council_republic
        }
        has_technology_researched = socialism
        any_scope_state = {
            is_in_geographic_region = geographic_region_atlantic_coast
            has_building = building_coal_mine
        }
        NOR = {
            AND = {
                has_law_or_variant = law_type:law_regulatory_bodies
                institution_investment_level = {
                    institution = institution_workplace_safety
                    value >= 5
                }
            }
            AND = {
                has_law_or_variant = law_type:law_worker_protections
                institution_investment_level = {
                    institution = institution_workplace_safety
                    value >= 3
                }
            }
            #Corporatized Unions variant
            AND = {
                has_law_or_variant = law_type:law_corporatized_unions
                has_institution = institution_workplace_safety
            }
        }
        ig:ig_trade_unions ?= {
            ig_counts_as_marginal = no
            ig_clout >= 0.1
        }
        ig:ig_industrialists ?= {
            ig_counts_as_marginal = no
        }
    }

    possible = {
        has_technology_researched = political_agitation
        any_scope_state = {
            has_building = building_coal_mine
            is_incorporated = yes
            any_scope_pop = {
                pop_employment_building = building_coal_mine
                OR = {
                    is_pop_type = laborers
                    is_pop_type = machinists
                }
                OR = {
                    pop_radical_fraction >= 0.15
                    standard_of_living < 14
                }
            }
        }
        ig:ig_trade_unions ?= {
            ig_counts_as_marginal = no
        }
        NOR = {
            has_law_or_variant = law_type:law_guild_system
            has_law_or_variant = law_type:law_combination_acts
            has_law_or_variant = law_type:law_corporatized_unions
            has_law_or_variant = law_type:law_factory_councils
        }
    }

    immediate = {
        scope:journal_entry = {
            add_com_dlc_icon = {
                dlc = dlc_usfp
            }
        }
        if = {
            limit = {
                NOT = {
                    any_political_movement = {
                        is_political_movement_type = movement_labor
                    }
                }
            }
            create_political_movement = {
                type = movement_labor
            }
        }
        every_scope_state = {
            limit = {
                has_building = building_coal_mine
                is_incorporated = yes
            }
            root = {
                add_to_variable_list = {
                    name = usfp_coal_wars_states
                    target = prev
                }
            }
        }
        every_in_list = {
            list = usfp_coal_wars_states
            set_variable = {
                name = usfp_coal_wars_state_unionization
                value = 0
            }
        }
        ordered_in_list = {
            variable = usfp_coal_wars_states
            order_by = "ig_state_pol_strength_share(root.ig:ig_trade_unions)"
            root = {
                set_variable = {
                    name = usfp_coal_wars_umwa_target
                    value = prev
                }
            }
            change_variable = {
                name = usfp_coal_wars_state_unionization
                add = 50
            }
        }
        trigger_event = usfp_coal_wars.1
    }

    status_desc = {
        first_valid = {
            triggered_desc = {
                trigger = {
                    scope:journal_entry = {
                        has_modifier = usfp_coal_wars_great_coal_strike
                    }
                }
                desc = usfp_coal_wars_great_coal_strike_ongoing
            }
            triggered_desc = {
                trigger = {
                    any_in_list = {
                        variable = usfp_coal_wars_states
                        count = all
                        has_variable = usfp_state_unionized
                    }
                }
                desc = usfp_coal_wars_all_states_unionized
            }
            triggered_desc = {
                trigger = {
                    has_variable = usfp_coal_wars_umwa_target
                }
                desc = usfp_coal_wars_umwa_organizing_state
            }
        }
    }

    on_monthly_pulse = {
        effect = {
            every_in_list = {
                list = usfp_coal_wars_states
                if = {
                    limit = {
                        var:usfp_coal_wars_state_unionization < 100
                    }
                    change_variable = {
                        name = usfp_coal_wars_state_unionization
                        add = usfp_coal_wars_unionization_progress
                    }
                }
                if = {
                    limit = {
                        var:usfp_coal_wars_state_unionization >= 100
                        this = root.var:usfp_coal_wars_umwa_target
                        NOT = {
                            is_target_in_variable_list = {
                                name = usfp_coal_wars_unionized_states
                                target = this
                            }
                        }
                    }
                    root = {
                        add_to_variable_list = {
                            name = usfp_coal_wars_unionized_states
                            target = prev
                        }
                    }
                    random_scope_building = {
                        limit = {
                            is_building_type = building_coal_mine
                        }
                        add_modifier = {
                            name = usfp_coal_wars_union_established_modifier
                        }
                    }
                    ordered_in_list = {
                        list = usfp_coal_wars_states
                        order_by = var:usfp_coal_wars_state_unionization
                        root = {
                            set_variable = {
                                name = usfp_coal_wars_umwa_target
                                value = prev
                            }
                        }
                    }
                    set_variable = usfp_state_unionized
                    save_scope_as = new_union_state
                    post_notification = usfp_coal_wars_state_unionized
                }
            }
        }
    }

    #workplace safety and the end of company towns
    complete = {
        #repeal lochner
        active_law:lawgroup_labor_rights = {
            NOT = {
                has_amendment = amendment_type:amendment_usfp_right_to_contract
            }
        }
        OR = {
            AND = {
                has_law_or_variant = law_type:law_regulatory_bodies
                institution_investment_level = {
                    institution = institution_workplace_safety
                    value >= 5
                }
            }
            AND = {
                has_law_or_variant = law_type:law_worker_protections
                institution_investment_level = {
                    institution = institution_workplace_safety
                    value >= 3
                }
            }
            #Corporatized Unions variant
            AND = {
                has_law_or_variant = law_type:law_corporatized_unions
                has_institution = institution_workplace_safety
            }
        }
    }


    on_complete = {
        trigger_event = usfp_coal_wars.7
    }

    #Crushing of the strikes and the TUs
    fail = {
        custom_tooltip = {
            text = progress_bar_zero
            scope:journal_entry = {
                scripted_bar_progress = {
                    name = usfp_coal_wars_union_leverage_bar
                    value <= 0
                }
            }
        }
    }

    on_fail = {
        trigger_event = usfp_coal_wars.8
    }

    #cleanup unionization modifiers

    #Council Republic
    invalid = {
        has_radical_leftist_economy = yes
    }


    should_be_pinned_by_default_uninvolved_or_context = yes

    weight = 100

    can_revolution_inherit = yes
}
