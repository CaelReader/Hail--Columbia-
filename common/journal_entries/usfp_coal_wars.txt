#Coal Wars
usfp_je_coal_wars = {
    icon = gfx/interface/icons/goods_icons/coal.dds
    group = je_group_historical_content
    dlc = dlc_usfp
    activation_description = { always = yes }

    scripted_progress_bar = usfp_coal_wars_union_leverage_bar

    modifiers_while_active = {
        usfp_coal_wars_ongoing_strikes
    }

    is_shown_when_inactive = {
        c:USA ?= root
        NOT = {
            has_law_or_variant = law_type:law_council_republic
        }
        has_technology_researched = socialism
        any_scope_state = {
            is_in_geographic_region = geographic_region_atlantic_coast
            has_building = building_coal_mine
        }
        NOR = {
            AND = {
                has_law_or_variant = law_type:law_regulatory_bodies
                institution_investment_level = {
                    institution = institution_workplace_safety
                    value >= 5
                }
            }
            AND = {
                has_law_or_variant = law_type:law_worker_protections
                institution_investment_level = {
                    institution = institution_workplace_safety
                    value >= 3
                }
            }
            #Corporatized Unions variant
            AND = {
                has_law_or_variant = law_type:law_corporatized_unions
                has_institution = institution_workplace_safety
            }
        }
        ig:ig_trade_unions ?= {
            ig_counts_as_marginal = no
            ig_clout >= 0.1
        }
        ig:ig_industrialists ?= {
            ig_counts_as_marginal = no
        }
    }

    possible = {
        has_technology_researched = political_agitation
        any_scope_state = {
            has_building = building_coal_mine
            is_incorporated = yes
            any_scope_pop = {
                pop_employment_building = building_coal_mine
                OR = {
                    is_pop_type = laborers
                    is_pop_type = machinists
                }
                OR = {
                    pop_radical_fraction >= 0.15
                    standard_of_living < 14
                }
            }
        }
        ig:ig_trade_unions ?= {
            ig_counts_as_marginal = no
        }
        NOR = {
            has_law_or_variant = law_type:law_guild_system
            has_law_or_variant = law_type:law_combination_acts
            has_law_or_variant = law_type:law_corporatized_unions
            has_law_or_variant = law_type:law_factory_councils
        }
    }

    immediate = {
        if = {
            limit = {
                NOT = {
                    any_political_movement = {
                        is_political_movement_type = movement_labor
                    }
                }
            }
            create_political_movement = {
                type = movement_labor
            }
        }
        every_scope_state = {
            limit = {
                has_building = building_coal_mine
                is_incorporated = yes
            }
            root = {
                add_to_variable_list = {
                    name = usfp_coal_wars_states
                    target = prev
                }
            }
        }
        every_in_list = {
            variable = usfp_coal_wars_states
            set_variable = {
                name = usfp_coal_wars_state_unionization
                value = 0
            }
        }
        ordered_in_list = {
            variable = usfp_coal_wars_states
            order_by = "ig_state_pol_strength_share(root.ig:ig_trade_unions)"
            position = 0
            root = {
                set_variable = {
                    name = usfp_coal_wars_umwa_target
                    value = prev
                }
            }
            change_variable = {
                name = usfp_coal_wars_state_unionization
                add = 50
            }
        }
        trigger_event = usfp_coal_wars.1
    }

    status_desc = {
        first_valid = {
            triggered_desc = {
                trigger = {
                    scope:journal_entry = {
                        has_modifier = usfp_coal_wars_great_coal_strike
                    }
                }
                desc = usfp_coal_wars_great_coal_strike_ongoing
            }
            triggered_desc = {
                trigger = {
                    any_in_list = {
                        variable = usfp_coal_wars_states
                        count = all
                        has_variable = usfp_coal_wars_state_unionized
                    }
                }
                desc = usfp_coal_wars_all_states_unionized
            }
            triggered_desc = {
                trigger = {
                    has_variable = usfp_coal_wars_umwa_target
                }
                desc = usfp_coal_wars_umwa_organizing_state
            }
        }
    }

    on_monthly_pulse = {
        effect = {
            every_in_list = {
                variable = usfp_coal_wars_states
                if = {
                    limit = {
                        var:usfp_coal_wars_state_unionization < 100
                    }
                    set_variable = {
                        name = usfp_coal_wars_state_unionization
                        value = {
                            value = var:usfp_coal_wars_state_unionization
                            add = usfp_coal_wars_unionization_progress
                            min = 0
                            max = 100
                        }
                    }
                }
                if = {
                    limit = {
                        var:usfp_coal_wars_state_unionization >= 100
                        this = root.var:usfp_coal_wars_umwa_target
                        NOT = {
                            is_target_in_variable_list = {
                                name = usfp_coal_wars_unionized_states
                                target = this
                            }
                        }
                    }
                    root = {
                        add_to_variable_list = {
                            name = usfp_coal_wars_unionized_states
                            target = prev
                        }
                    }
                    random_scope_building = {
                        limit = {
                            is_building_type = building_coal_mine
                        }
                        add_modifier = {
                            name = usfp_coal_wars_union_established_modifier
                        }
                    }
                    set_variable = usfp_coal_wars_state_unionized
                    ordered_in_list = {
                        variable = usfp_coal_wars_states
                        order_by = var:usfp_coal_wars_state_unionization
                        position = 0
                        limit = {
                            NOT = {
                                has_variable = usfp_coal_wars_state_unionized
                            }
                        }
                        root = {
                            set_variable = {
                                name = usfp_coal_wars_umwa_target
                                value = prev
                            }
                        }
                    }
                    save_scope_as = new_union_state
                    owner = {
                        post_notification = usfp_coal_wars_state_unionized
                    }
                }
            }
        }
        random_events = {
            chance_of_no_event = 50
            100 = usfp_coal_wars.2
            100 = usfp_coal_wars.3
            100 = usfp_coal_wars.4
            100 = usfp_coal_wars.5
            100 = usfp_coal_wars.9
        }
        events = {
            usfp_coal_wars.6
            usfp_coal_wars.10
        }
    }

    map_color = {
        if = {
            limit = {
                owner = {
                    is_target_in_variable_list = { name = usfp_coal_wars_states target = root }
                }
            }
            value = "grey"
        }
        if = {
            limit = {
                has_variable = usfp_coal_wars_state_unionized
            }
            value = "yellow"
        }
        else_if = {
            limit = {
                has_variable = usfp_coal_wars_state_unionization
            }
            lerp = {
                min_color = "grey" # color
                max_color = "red" # color
                factor = {
                    value = this.var:usfp_coal_wars_state_unionization
                    divide = 100
                }
            }
        }
    }

    map_color_second = {
        if = {
            limit = {
                this = owner.var:usfp_coal_wars_umwa_target
            }
            value = "yellow"
        }
    }

    map_tooltip = {
        if = {
            limit = {
                owner = {
                    has_variable = usfp_coal_wars_umwa_target
                }
                this = owner.var:usfp_coal_wars_umwa_target
            }
            custom_tooltip_no_bullet = "coal_wars_umwa_target_state"
        }
        else_if = {
            limit = {
                has_building = building_coal_mine
                is_incorporated = yes
            }
            custom_tooltip_no_bullet = "coal_wars_valid_state"
        } else_if = {
            limit = {
                has_building = building_coal_mine
                is_incorporated = no
            }
            custom_tooltip_no_bullet = "coal_wars_unincorporated_state"
        }
        if = {
            limit = {
                has_variable = usfp_coal_wars_state_unionized
            }
            custom_tooltip_no_bullet = "coal_wars_state_unionized"
        }
        else_if = {
            limit = {
                has_variable = usfp_coal_wars_state_unionization
            }
            custom_tooltip_no_bullet = "coal_wars_state_unionization_level"
            custom_tooltip_no_bullet = "coal_wars_state_unionization_progress"
        }
    }

    #workplace safety and the end of company towns
    outcome = {
        header = je_standard_completion_header
        effect_header = je_standard_on_completion_header
        color = "gold_positive"
        conclusion_desc = je_standard_completion_conclusion_desc
        flags = { historical }
        trigger = {
            #repeal lochner
            active_law:lawgroup_labor_rights = {
                NOT = {
                    has_amendment = amendment_type:amendment_usfp_right_to_contract
                }
            }
            NOT = { has_law_or_variant = law_type:law_child_labor_allowed }
            NOT = { has_law_or_variant = law_type:law_combination_acts }
            OR = {
                AND = {
                    has_law_or_variant = law_type:law_regulatory_bodies
                    institution_investment_level = {
                        institution = institution_workplace_safety
                        value >= 5
                    }
                }
                AND = {
                    has_law_or_variant = law_type:law_worker_protections
                    institution_investment_level = {
                        institution = institution_workplace_safety
                        value >= 3
                    }
                }
                #Corporatized Unions variant
                AND = {
                    has_law_or_variant = law_type:law_corporatized_unions
                    has_institution = institution_workplace_safety
                }
            }
        }
        effect = {
            trigger_event = usfp_coal_wars.7
            usfp_coal_wars_cleanup = yes
        }
        event_outcome_effect_desc = {
            effect = {
                execute_event_option = {
                    event = usfp_coal_wars.7
                    option = 0
                }
            }
        }
    }

    #Crushing of the strikes and the TUs
    outcome = {
        header = je_standard_failure_header
        effect_header = je_standard_on_failure_header
        color = "blue_negative"
        conclusion_desc = je_standard_failure_conclusion_desc
        trigger = {
            custom_tooltip = {
                text = progress_bar_zero
                scope:journal_entry = {
                    scripted_bar_progress = {
                        name = usfp_coal_wars_union_leverage_bar
                        value <= 0
                    }
                }
            }
        }
        effect = {
            trigger_event = usfp_coal_wars.8
            usfp_coal_wars_cleanup = yes
        }
        event_outcome_effect_desc = {
            effect = {
                execute_event_option = {
                    event = usfp_coal_wars.8
                    option = 0
                }
            }
        }
    }


    outcome = {
        flags = { hidden invalid_outcome }
        conclusion_desc = je_standard_completion_conclusion_desc
        trigger = {
            OR = {
                has_radical_leftist_economy = yes
                NOT = {
                    any_scope_state = {
                        has_building = building_coal_mine
                        is_incorporated = yes
                    }
                }
            }
        }
        effect = {
            usfp_coal_wars_cleanup = yes
        }
    }


    should_be_pinned_by_default_uninvolved_or_context = yes

    weight = 100

    can_revolution_inherit = yes
}
