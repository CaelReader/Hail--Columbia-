#from USCWF and VTM
je_oregon = {
	icon = "gfx/interface/icons/event_icons/event_map.dds"

	group = je_group_us_expansion_global

    should_show_when_not_involved = {
        OR = {
            this = c:USA
            this = c:GBR
            is_in_geographic_region = geographic_region_canada
            any_scope_state = {
                OR = {
                    state_region = s:STATE_WASHINGTON
                    state_region = s:STATE_OREGON
                    state_region = s:STATE_IDAHO
                    state_region = s:STATE_BRITISH_COLUMBIA
                }
            }
        }
    }

    should_be_involved = {
        OR = {
            this = c:USA
            this = c:GBR
            AND = {
                is_subject_of = c:GBR
                any_scope_state = {
                    OR = {
                        state_region = s:STATE_WASHINGTON
                        state_region = s:STATE_OREGON
                        state_region = s:STATE_IDAHO
                        state_region = s:STATE_BRITISH_COLUMBIA
                    }
                }
            }
        }
    }

    should_be_pinned_by_default_involved = yes

    immediate = {
        c:ORG ?= {
            save_scope_as = oregon_holder_scope
        }
        c:GBR ?= {
            save_scope_as = gbr_scope
        }
        c:USA ?= {
            save_scope_as = usa_scope
        }
    }

	complete = {
        custom_tooltip = {
            text = oregon_tt
            has_global_variable = oregon_trail_mapped
        }
        c:GBR = {
            scope:usa_scope = {
                relations:prev >= relations_threshold:cordial
            }
        }
        c:USA = {
            has_interest_marker_in_region = sr:region_pacific_coast
            OR = {
                OR = {
                    usfp_usa_has_state_in_montana_wyoming_california = yes
                    usfp_usa_or_subject_has_state_in_all_southern_oregon_territory = yes
                }
                AND = {
                    c:GBR = {
                        scope:usa_scope = {
                            relations:prev >= relations_threshold:amicable
                        }
                    }
                    s:STATE_CALIFORNIA = {
                        any_scope_state = {
                            owner = {
                                has_war_with = c:USA
                            }
                        }
                    }
                }
            }
        }
	}

	on_complete = {
		if = {
			limit = {
				s:STATE_WASHINGTON = {
					any_scope_state = {
						owner = {
							OR = {
								c:GBR = THIS
								is_subject_of = c:GBR
							}
						}
					}
				}
				s:STATE_OREGON = {
					any_scope_state = {
						owner = {
							OR = {
								c:GBR = THIS
								is_subject_of = c:GBR
							}
						}
					}
				}
				s:STATE_IDAHO = {
					any_scope_state = {
						owner = {
							OR = {
								c:GBR = THIS
								is_subject_of = c:GBR
							}
						}
					}
				}
			}
			c:GBR ?= { trigger_event = { id = oregon.2 } }
		}
		else = {
			s:STATE_BRITISH_COLUMBIA = {
				random_scope_state = {
					limit = {
						owner = {
							is_subject_of = c:GBR
						}
					}
					owner = {
						save_scope_as = oregon_holder_scope
					}
				}
			}
			s:STATE_IDAHO = {
				add_claim = c:USA
				remove_claim = scope:oregon_holder_scope
			}
			s:STATE_WASHINGTON = {
				add_claim = c:USA
				remove_claim = scope:oregon_holder_scope
			}
			s:STATE_OREGON = {
				add_claim = c:USA
				remove_claim = scope:oregon_holder_scope
			}
			add_modifier = {
				name = oregon_resolved
				days = normal_modifier_time
			}
		}
        c:USA = {
            if = {
                limit = {
                    has_journal_entry = usfp_manifest_destiny_main
                }
                usfp_manifest_less_destiny_high_effect = yes
            }
        }
	}

    event_outcome_completed_effect_desc = {
        effect = {
            execute_event_option = {
                event = oregon.2
                option = 0
            }
        }
    }

	fail = {
		OR = {
            c:GBR = {
                has_diplomats_expelled = c:USA
            }
            c:GBR = {
                has_war_with = c:USA
            }
			c:GBR = {
                scope:usa_scope = {
                    relations:prev <= relations_threshold:poor
                }
			}
            usfp_gbr_or_subject_has_state_in_southern_oregon_territory = no
		}
	}
	
	on_fail = {
			s:STATE_OREGON = {
				add_claim = c:USA
			}
			s:STATE_IDAHO = {
				add_claim = c:USA
			}
			s:STATE_WASHINGTON = {
				add_claim = c:USA
			}
            c:USA = {
                add_journal_entry = {
                    type = je_conquer_oregon
                }
                if = {
                    limit = {
                        has_journal_entry = usfp_manifest_destiny_main
                    }
                    usfp_manifest_more_destiny_effect = yes
                }
            }
	}

    invalid = {
        OR = {
            NOT = {
                exists = c:USA
            }
            NOT = {
                exists = c:GBR
            }
        }
    }

	weight = 100
}

je_conquer_oregon = {
	icon = "gfx/interface/icons/event_icons/event_military.dds"

	group = je_group_usa_manifest_destiny

	scripted_button = usfp_demand_bc_button 

	complete = {
		c:USA ?= {
			owns_entire_state_region = STATE_WASHINGTON
			owns_entire_state_region = STATE_IDAHO
			owns_entire_state_region = STATE_OREGON
		}
	}

	fail = {
		NOT = {
			has_claim = s:STATE_WASHINGTON
			has_claim = s:STATE_IDAHO
			has_claim = s:STATE_OREGON
		}
	}
	

	weight = 100
	should_be_pinned_by_default_uninvolved_or_context = yes
	transferable = no
}
