usfp_usa_labor.1 = {
	type = country_event

	icon = "gfx/interface/icons/invention_icons/anarchy.dds"
	event_image = {
		video = "votp_agitator_speaking"
	}

	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"

	placement = scope:illinois_scope

	title = usfp_usa_labor.1.t
	desc = usfp_usa_labor.1.d
	flavor = usfp_usa_labor.1.f
	dlc = dlc_usfp
	duration = 3

	trigger = {
		exists = c:USA
		c:USA = root
		NOT = {
			has_global_variable = iww_founded
		}
		game_date > 1905.6.1
		has_technology_researched = anarchism
		has_technology_researched = labor_movement
		has_technology_researched = socialism
		ig:ig_trade_unions = {
			ig_counts_as_marginal = no
		}
		any_scope_state = {
			state_region = s:STATE_ILLINOIS
		}
		NOT = {
			has_law_or_variant = law_type:law_cooperative_ownership
		}
	}

	immediate = {
		set_global_variable = iww_founded
		s:STATE_ILLINOIS.region_state:USA = {
			save_scope_as = illinois_scope
		}
		create_character = {
			template = usfp_bill_haywood_character_template
			on_created = {
				set_variable = is_bill_haywood	
			}
		}
		random_scope_character = {
			limit = {
				has_variable = is_bill_haywood	
			}
			save_scope_as = haywood_scope
		}
		ig:ig_trade_unions = {
			save_scope_as = trade_unions_scope
		}
	}

	#Solidarity
	option = {
		name = usfp_usa_labor.1.a
		default_option = yes
		ig:ig_trade_unions = {
			set_ig_trait = ig_trait:usfp_ig_trait_direct_action
			add_modifier = {
				name = usfp_iww_ig
				years = 15
			}
		}
		# scope:haywood_scope = {
		# 	set_as_interest_group_leader = yes
		# }
		add_modifier = {
			name = usfp_labor_agitation
			years = 1
		}
	}

	#Damn Radicals
	option = {
		name = usfp_usa_labor.1.b
		ig:ig_trade_unions = {
			add_modifier = {
				name = usfp_iww_ig
				years = 15
				multiplier = 0.5
			}
		}
		add_radicals = {
			value = medium_radicals
			strata = lower
		}
		add_modifier = {
			name = usfp_labor_agitation
			years = 2
		}
	}
}

	desc = usfp_usa_labor.2.d
	flavor = usfp_usa_labor.2.f
	dlc = dlc_usfp
	duration = 3

	trigger = {
		exists = c:USA
		c:USA = root
		game_date > 1886.5.1
		NOT = {
			has_global_variable = had_haymarket_affair
		}
		ig:ig_trade_unions = {
			ig_counts_as_marginal = no
		}
		any_scope_state = {

				value > 0.05
			}
		}

			has_law_or_variant = law_type:law_no_police

	immediate = {
		set_global_variable = had_haymarket_affair
		s:STATE_ILLINOIS.region_state:USA = {
			save_scope_as = illinois_scope
		}
	}

	#Prosecute!
	option = {
		name = usfp_usa_labor.2.a
		default_option = yes
		add_radicals = {
			value = small_radicals
			interest_group = ig:ig_trade_unions
		}
		ig:ig_industrialists = {
			add_modifier = {
				name = usfp_prosecuted_anarchists_positive
				years = 2
			}
		}
		ig:ig_petty_bourgeoisie = {
			add_modifier = {
				name = usfp_prosecuted_anarchists_positive
				years = 2
			}
		}
		if = {
			limit = { 
				ig:ig_trade_unions = {
					leader = {
						NOR = {
							has_ideology = ideology:ideology_vanguardist
							has_ideology = ideology:ideology_state_socialist
						}
					}
				}
			}
			ig:ig_trade_unions = {
				add_modifier = {
					name = usfp_prosecuted_anarchists_negative
					years = 2
				}
			}
		}
		ig:ig_trade_unions = {
			add_modifier = {
				name = usfp_international_workers_day
				years = 5
			}
		}
	}

	#Let them go
	option = {
		name = usfp_usa_labor.2.b

			}
