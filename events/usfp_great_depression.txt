#Hoovervilles
usfp_great_depression.6 = {
    type = country_event
    placement = root
    event_image = {
        video = "unspecific_vandalized_storefront"
    }

    on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"

    icon = "gfx/interface/icons/invention_icons/urbanization.dds"

    title = usfp_great_depression.6.t
    desc = usfp_great_depression.6.d
    flavor = usfp_great_depression.6.f
    dlc = dlc_usfp
    duration = 3
    cooldown = { days = never_fire_again_modifier_time } #Once

    trigger = {
        any_political_movement = {
            movement_is_socialist = yes
        }
        any_scope_state = {
            state_unemployment_rate > 0.10
            count >= 3
        }
    }

    immediate = {
        ordered_political_movement = {
            limit = {
                movement_is_socialist = yes
            }
            position = 0
            order_by = political_movement_support
            save_scope_as = socialist_movement
        }
    }

    #sweep
    option = {
        name = usfp_great_depression.6.a
        default_option = yes
        add_radicals = {
            strata = lower
            value = medium_radicals
        }
    }

    #agitate
    option = {
        name = usfp_great_depression.6.b
        scope:socialist_movement = {
            add_modifier = {
                name = usfp_hooverville_agitation_modifier
                years = 5
            }
        }
    }

    after = {
        every_scope_state = {
            custom = hooverville_states
            limit = {
                state_unemployment_rate > 0.10
            }
        }
    }
}
