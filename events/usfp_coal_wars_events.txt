        save_scope_as = coal_mine_state

    option = {
        name = usfp_coal_wars.1.a
        default_option = yes

    }
}

# Mother Jones Spawn
usfp_coal_wars.2 = {
    type = country_event
    placement = root

    on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"

    icon = "gfx/interface/icons/event_icons/event_protest.dds"
    gui_window = event_window_1char_tabloid
    left_icon = scope:mother_jones_scope
    on_opened_soundeffect = "event:/SFX/Events/misc/1Character_Banner"

    title = usfp_coal_wars.2.t
    desc = usfp_coal_wars.2.d
    flavor = usfp_coal_wars.2.f
    dlc = dlc_usfp
    duration = 3
    cooldown = { days = never_fire_again_modifier_time } #Once

    trigger = {
        NOT = {
            any_scope_character = {
                has_template = usfp_mother_jones
            }
        }
        year >= 1871
    }

    immediate = {
        create_character = {
            template = usfp_mother_jones
            is_agitator = yes
            save_scope_as = mother_jones_scope
        }
    }

    option = {
        name = usfp_coal_wars.2.a
        default_option = yes

    }
}
