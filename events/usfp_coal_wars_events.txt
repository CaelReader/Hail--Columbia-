# Great Coal Strike
usfp_coal_wars.6 = {
    type = country_event
    placement = root
    event_image = {
        video = "gfx/event_pictures/votp_agitator_speaking.bk2"
    }

    on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"

    icon = "gfx/interface/icons/event_icons/event_protest.dds"

    title = usfp_coal_wars.6.t
    desc = usfp_coal_wars.6.d
    flavor = usfp_coal_wars.6.f
    dlc = dlc_usfp
    duration = 3
    cooldown = { days = never_fire_again_modifier_time } #Once

    trigger = {

    }

    immediate = {

    }

    #let them starve
    option = {
        name = usfp_coal_wars.6.a
        default_option = yes
        add_radicals = {
            value = small_radicals
            interest_group = ig:ig_trade_unions
        }
    }

    #immediate negotiations
    option = {
        name = usfp_coal_wars.6.b
        highlighted_option = yes
        trigger = {
            OR = {
                is_enacting_law = law_type:law_worker_protections
                has_law_or_variant = law_type:law_worker_protections
            }
        }
        show_as_unavailable = { always = yes }
        if = {
            limit = {
                is_enacting_law = law_type:law_worker_protections
            }
            add_enactment_phase = 1
        } else_if = {
            limit = {
                has_law_or_variant = law_type:law_worker_protections
                institution_investment_level = {
                    institution = institution_workplace_safety
                    value < 5
                }
            }
            change_institution_investment_level = {
                institution = institution_workplace_safety
                investment = 1
            }
        }
    }

    after = {
        custom_tooltip = usfp_coal_wars_strike_will_drain_leverage
        je:usfp_je_coal_wars = {
            add_modifier = usfp_coal_wars_great_coal_strike
        }
    }
}

#Unions throw in with socialist revolution
usfp_coal_wars.10 = {
    type = country_event
    placement = scope:kentucky_scope
    event_image = {
        video = "gfx/event_pictures/votp_agitator_speaking.bk2"
    }

    on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"

    icon = "gfx/interface/icons/event_icons/event_protest.dds"

    title = usfp_coal_wars.10.t
    desc = usfp_coal_wars.10.d
    flavor = usfp_coal_wars.10.f
    dlc = dlc_usfp
    duration = 3
    cooldown = { days = normal_modifier_time }

    trigger = {
        any_political_movement = {
            OR = {
                movement_is_socialist = yes
                is_political_movement_type = movement_labor
            }
            is_insurrectionary = yes
            civil_war ?= {
                civil_war_progress >= 0.50
            }
        }
    }

    immediate = {
        ordered_political_movement = {
            limit = {
                OR = {
                    movement_is_socialist = yes
                    is_political_movement_type = movement_labor
                }
                is_insurrectionary = yes
                civil_war ?= {
                    civil_war_progress >= 0.50
                }
            }
            order_by = political_movement_support
            save_scope_as = revolutionary_movement
        }
    }

    #viva la revolution
    option = {
        name = usfp_coal_wars.10.a
        default_option = yes
        every_scope_state = {
            custom = coal_wars_every_unionized_state
            limit = {
                has_variable = usfp_state_unionized
            }
            add_radicals_in_state = {
                value = large_radicals
                interest_group = root.ig:ig_trade_unions
            }
        }
    }

    #reinforce patriotic labor
    option = {
        name = usfp_coal_wars.10.b
        highlighted_option = yes
        trigger = {
            ig:ig_trade_unions ?= {
                leader = {
                    has_socialist_ideology = no
                }
            }
        }
        show_as_unavailable = { always = yes }
        every_scope_state = {
            custom = coal_wars_every_unionized_state
            limit = {
                has_variable = usfp_state_unionized
            }
            add_radicals_in_state = {
                value = medium_radicals
                pop_type = laborers
            }
        }
        add_loyalists = {
            interest_group = root.ig:ig_trade_unions
            value = small_radicals
        }
    }

}
