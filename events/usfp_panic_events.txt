#Panic of 1837
usfp_panics.1 = {
	type = country_event
	placement = s:STATE_NEW_YORK.region_state:USA
	duration = 3
	cooldown = { days = never_fire_again_modifier_time } #Once

	title = usfp_panics.1.t
	desc = usfp_panics.1.d
	flavor = usfp_panics.1.f

	dlc = dlc_usfp
	event_image = {
		video = "unspecific_factory_closed"
	}
	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"
	icon = "gfx/interface/icons/invention_icons/banking.dds"


	#uh oh
	option = {
		name = usfp_panics.1.a
		default_option = yes
		add_journal_entry = {
			type = je_usfp_panic_1837
		}
		show_as_tooltip = {
			add_modifier = usfp_panic_1837_modifier
		}
		random_scope_state = {
			limit = {
				OR = {
					state_region = s:STATE_ARKANSAS
					state_region = s:STATE_ILLINOIS
				}
				NOT = {
					has_modifier = usfp_panic_state_default_modifier
				}
			}
			add_modifier = {
				name = usfp_panic_state_default_modifier
				years = 4
			}
		}
		random_scope_state = {
			limit = {
				OR = {
					state_region = s:STATE_INDIANA
					state_region = s:STATE_MARYLAND
				}
				NOT = {
					has_modifier = usfp_panic_state_default_modifier
				}
			}
			add_modifier = {
				name = usfp_panic_state_default_modifier
				years = 4
			}
		}
		random_scope_state = {
			limit = {
				OR = {
					state_region = s:STATE_LOUISIANA
					state_region = s:STATE_MISSISSIPPI
				}
				NOT = {
					has_modifier = usfp_panic_state_default_modifier
				}
			}
			add_modifier = {
				name = usfp_panic_state_default_modifier
				years = 4
			}
		}
		random_scope_state = {
			limit = {
				state_region = s:STATE_PENNSYLVANIA
				NOT = {
					has_modifier = usfp_panic_state_default_modifier
				}
			}
			add_modifier = {
				name = usfp_panic_state_default_modifier
				years = 4
			}
		}
		ruler = {
			add_modifier = {
				name = usfp_blamed_for_panic
				years = 5
			}
		}
	}

	#good thing we have a central bank right
	option = {
		name = usfp_panics.1.b
		trigger = {
			has_law_or_variant = law_type:law_interventionism
		}
		show_as_unavailable = {
			always = yes
		}
		highlighted_option = yes
		add_journal_entry = {
			type = je_usfp_panic_1837
		}
		show_as_tooltip = {
			add_modifier = {
				name = usfp_panic_1837_modifier
				multiplier = 0.5
			}
		}
		add_treasury = -500000
	}

}

#Panic of 1873
usfp_panics.2 = {
	type = country_event
	placement = s:STATE_NEW_YORK.region_state:USA
	duration = 3
	cooldown = { days = never_fire_again_modifier_time } #Once

	title = usfp_panics.2.t
	desc = usfp_panics.2.d
	flavor = usfp_panics.2.f

	dlc = dlc_usfp
	event_image = {
		video = "unspecific_factory_closed"
	}
	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"
	icon = "gfx/interface/icons/invention_icons/banking.dds"


	#uh oh
	option = {
		name = {
			trigger = {
				has_variable = railroad_bubble_investment_low
			}
			text = usfp_panics.2.a.low
		}
		name = usfp_panics.2.a
		default_option = yes
		if = {
			limit = {
				has_variable = railroad_bubble_investment_low
			}
			add_modifier = {
				name = usfp_panic_1873_modifier
				years = 5
				is_decaying = yes
				multiplier = 0.5
			}
			add_investment_pool = {
				value = money_amount_multiplier_very_large
				multiply = -1
			}
		} else = {
			add_modifier = {
				name = usfp_panic_1873_modifier
				years = 5
				is_decaying = yes
			}
			add_investment_pool = {
				value = money_amount_multiplier_very_large
				multiply = -2
			}
		}


	}

	#backstop failing firms
	option = {
		name = usfp_panics.2.b
		trigger = {
			has_law_or_variant = law_type:law_interventionism
		}
		show_as_unavailable = {
			always = yes
		}
		highlighted_option = yes
		if = {
			limit = {
				has_variable = railroad_bubble_investment_low
			}
			add_modifier = {
				name = usfp_panic_1873_modifier
				years = 5
				is_decaying = yes
				multiplier = 0.40
			}
			add_investment_pool = {
				value = money_amount_multiplier_very_large
				multiply = -0.5
			}
		} else = {
			add_modifier = {
				name = usfp_panic_1873_modifier
				years = 5
				multiplier = 0.80
				is_decaying = yes
			}
			add_investment_pool = {
				value = money_amount_multiplier_very_large
				multiply = -1
			}
		}
		add_treasury = {
			value = money_amount_multiplier_very_large
			multiply = -1
		}
	}

	after = {
		ruler = {
			add_modifier = {
				name = usfp_blamed_for_panic
				years = 5
			}
		}
	}

}
