namespace = usfp_dust_bowl

#Great Plow Up
usfp_dust_bowl.1 = {
    type = country_event
    placement = scope:plow_up_state
    event_image = {
        video = "ip4_colonial_exploitation_going_well"
    }

    on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"

    icon = "gfx/interface/icons/invention_icons/mechanized_farming.dds"

    title = usfp_dust_bowl.1.t
    desc = usfp_dust_bowl.1.d
    flavor = usfp_dust_bowl.1.f
    dlc = dlc_usfp
    duration = 3
    cooldown = { days = never_fire_again_modifier_time } #Once

    trigger = {
        has_technology_researched = compression_ignition
        has_technology_researched = mechanized_farming
        any_scope_state = {
            is_in_geographic_region = geographic_region_southern_great_plains
            NOT = {
                state_region = {
                    has_harvest_condition = drought
                }
            }
            state_region = {
                OR = {
                    has_harvest_condition = optimal_sunlight
                    has_harvest_condition = moderate_rainfall
                }
            }
        }
        NOT = {
            has_global_variable = dust_bowl_happened
        }
    }

    immediate = {
        ordered_scope_state = {
            position = 0
            order_by = gdp
            limit = {
                is_in_geographic_region = geographic_region_southern_great_plains
            }
            save_scope_as = plow_up_state
        }
        set_global_variable = had_great_plow_up
    }

    #more wheat gooder
    option = {
        name = usfp_dust_bowl.1.a
        default_option = yes
        every_state_region_in_southern_plains = {
            add_arable_land = 5
        }
        every_scope_state = {
            limit = {
                is_in_geographic_region = geographic_region_southern_great_plains
            }
            add_modifier = {
                name = usfp_great_plow_up_state_modifier
                years = 10
            }
        }
    }

    #commercialized ag
    option = {
        name = usfp_dust_bowl.1.b
        show_as_unavailable = { always = yes }
        trigger = {
            has_law_or_variant = law_type:law_commercialized_agriculture
        }
        highlighted_option = yes
        every_state_region_in_southern_plains = {
            add_arable_land = 5
        }
        every_scope_state = {
            limit = {
                is_in_geographic_region = geographic_region_southern_great_plains
            }
            add_modifier = {
                name = usfp_great_plow_up_state_modifier
                years = 10
            }
            add_modifier = {
                name = usfp_great_plow_up_investment_modifier
                years = 10
            }
            add_loyalists_in_state = {
                pop_type = capitalists
                value = medium_radicals
            }
        }
    }

    #collectivized ag
    option = {
        name = usfp_dust_bowl.1.c
        trigger = {
            has_law_or_variant = law_type:law_collectivized_agriculture
        }
        highlighted_option = yes
        every_state_region_in_southern_plains = {
            add_arable_land = 5
        }
        every_scope_state = {
            limit = {
                is_in_geographic_region = geographic_region_southern_great_plains
            }
            add_modifier = {
                name = usfp_great_plow_up_state_modifier
                years = 10
            }
            add_modifier = {
                name = usfp_great_plow_up_collectives_modifier
                years = 10
            }
            add_loyalists_in_state = {
                pop_type = farmers
                value = medium_radicals
            }
        }
    }

}

usfp_dust_bowl.2 = {
    type = country_event
    placement = scope:dust_bowl_state
    event_image = {
        video = "asia_dead_cattle_poor_harvest"
    }

    on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"

    icon = "gfx/interface/icons/harvest_condition_icons/usfp_dust_storm.dds"

    title = usfp_dust_bowl.2.t
    desc = usfp_dust_bowl.2.d
    flavor = usfp_dust_bowl.2.f
    dlc = dlc_usfp
    duration = 3
    cooldown = { days = never_fire_again_modifier_time } #Once

    trigger = {
        #JE Start
    }

    immediate = {
        random_scope_state = {
            limit = {
                is_in_geographic_region = geographic_region_southern_great_plains
            }
            save_scope_as = dust_bowl_state
            state_region = {
                start_harvest_condition_with_params = {
                    type = usfp_dust_storm
                    intensity = 3
                    duration = 60
                }
            }
        }
        random_scope_state = {
            limit = {
                is_in_geographic_region = geographic_region_southern_great_plains
                NOT = {
                    this = scope:dust_bowl_state
                }
            }
            state_region = {
                start_harvest_condition_with_params = {
                    type = usfp_dust_storm
                    intensity = 3
                    duration = 60
                }
            }
        }
    }

    #oh dear
    option = {
        name = usfp_dust_bowl.2.a
        default_option = yes
        every_scope_state = {
            limit = {
                has_modifier = usfp_great_plow_up_state_modifier
            }
            remove_modifier = usfp_great_plow_up_state_modifier
        }
        custom_tooltip = dust_bowl_dust_storms_begin
        custom_tooltip = dust_bowl_dust_storms_arable_land_warning
    }


}

        }

        name = usfp_dust_bowl.6.b

                name = usfp_dust_bowl_resettlment_modifier
                years = 5

            add_modifier = {
